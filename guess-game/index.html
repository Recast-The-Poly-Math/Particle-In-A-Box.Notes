Escape the Infinity Loop

=== GUESS THE NUMBER ===

Guess a number from 1 to 10. The game gives you guess + secretNumber as a hint, then you make your final guess. Enter 0 as the first guess to escape the loop.

Think How to Escape the Infinity Loop!!!
Original C++ source 
#include <iostream>
#include <random>
using namespace std;

int main() {
    random_device rd;
    mt19937 generator(rd());
    uniform_int_distribution<int> randomNumber(1, 10);

    int secretNumber = randomNumber(generator);
    int guess = -1;
    int hint;
    int guess2;

    while (guess != 0) {
        cout << "Think How to Escape the Infinity Loop!!!" << endl;
        cout << "=== GUESS THE NUMBER ===\n";
        cout << "Guess a number from 1 to 10: ";

        cin >> guess;
        if (guess == 0) break;

        hint = guess + secretNumber;
        cout << endl << "My first output: " << hint << endl;

        cout << "Guess now!" << endl;
        cin >> guess2;

        if (guess2 == secretNumber)
            cout << "You win!\n";
        else
            cout << "You lose. The number was " << secretNumber << ".\n";

        secretNumber = randomNumber(generator);
    }

    return 0;
}
