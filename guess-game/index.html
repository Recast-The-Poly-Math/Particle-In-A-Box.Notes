Escape the Infinity Loop

=== GUESS THE NUMBER ===

Make a first guess from 1 to 10. The game gives you guess + secretNumber as a hint, then you make your final guess.

Think How to Escape the Infinity Loop!!!
Original C++ idea
#include <iostream>
#include <random>
using namespace std;

int main() {
    random_device rd;
    mt19937 generator(rd());
    uniform_int_distribution<int> randomNumber(1, 10);
    int secretNumber = randomNumber(generator);
    int guess, hint, guess2;

    cout << "Guess a number from 1 to 10: ";
    cin >> guess;
    hint = guess + secretNumber;
    cout << "My first output: " << hint << endl;
    cout << "Guess now: ";
    cin >> guess2;

    if (guess2 == secretNumber) cout << "You win!\n";
    else cout << "You lose. The number was " << secretNumber << ".\n";
    return 0;
}
